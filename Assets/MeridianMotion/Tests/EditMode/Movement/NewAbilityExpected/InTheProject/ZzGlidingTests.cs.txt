using System.Collections;
using MeridianMotion.Characters;
using MeridianMotion.Tests;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;

namespace Abilities.Tests
{
    /// <summary>
    /// Tests of Zz Gliding (made by New Ability). The checks every ability gets (it starts, turned off it doesn't,
    /// letting go, its events, saved and put back, at another size) come from AbilityFixture: as set up here,
    /// a press of Interact standing on a floor.
    /// Add the tests that are about Zz Gliding itself below, reading expected values from its settings
    /// (rig.MovementSetting&lt;float&gt;("zzGlidingDuration")), not from numbers of your own.
    /// </summary>
    [Category(TestArea.Abilities)]
    public class ZzGlidingTests : AbilityFixture<ZzGliding>
    {
        protected override IEnumerator SetUpToStart()
        {
            rig.Floor();
            rig.Track(new GameObject("ZzGlidingPoint")).AddComponent<ZzGlidingPoint>();
            // At the element, it starts at once: not settled first, so the kit sees it start.
            yield return SpawnCharacter(new Vector3(0f, 0.05f, 0f), settleSeconds: 0f);
        }

        protected override StepInputs InputsFor(int step) => step == 0 ? CharacterTestRig.Interact() : CharacterTestRig.Idle();

        protected override string OffSetting => "zzGlidingDuration";

        [UnityTest]
        public IEnumerator ZzGliding_LastsItsDuration()
        {
            yield return SetUpToStart();
            float duration = rig.MovementSetting<float>("zzGlidingDuration") * rig.Movement.DurationScale;
            int steps = 0;
            yield return rig.Run(CharacterTestRig.StepsFor(duration) + 5, InputsFor, _ => { if (rig.InState(ZzGliding.State)) { steps++; } });
            Assert.That(steps, Is.EqualTo(CharacterTestRig.StepsFor(duration)).Within(1), "Zz Gliding should last its duration.");
        }
    }
}
