using System.Collections.Generic;
using UnityEngine;

namespace {{ElementNamespace}}
{
    /// <summary>
    /// A place in the level where the {{Display}} ability starts (made by New Ability): a character within its
    /// ability's reach of one starts it. It knows nothing about the character (the level never does): the ability finds
    /// it through All. Replace it with what your ability uses in the level. If it gets state of its own that the
    /// character's movement depends on (as a rope's swing does), see docs/adding-level-content.md for the rules it keeps.
    /// </summary>
    [AddComponentMenu("{{MenuRoot}}/Levels/{{Display}} {{ElementDisplayTail}}")]
    public class {{Element}} : MonoBehaviour
    {
        private static readonly List<{{Element}}> active = new();

        /// <summary>Every enabled {{Element}}, for characters to look for one.</summary>
        public static IReadOnlyList<{{Element}}> All => active;

        private void OnEnable() => active.Add(this);

        private void OnDisable() => active.Remove(this);

        private void OnDrawGizmos()
        {
            Gizmos.color = new Color(0.3f, 0.8f, 1f, 0.8f);
            Gizmos.DrawWireSphere(transform.position, 0.25f);
        }
    }
}
