using System;
{{#levelElement}}
using MeridianMotion.Levels;
{{/levelElement}}
using UnityEngine;

namespace MeridianMotion.Characters
{
    /// <summary>
    /// {{Display}}: made by New Ability (Tools > Meridian Motion > New Ability…). As made, it's a small working
    /// ability to build on:
{{#takesOver}}
    /// it takes the character over and holds it where it is, hovering, for {{Display}} Duration, then lets go.
{{/takesOver}}
{{#onTopOfWalking}}
    /// on top of walking, it moves the character on the way it faces at {{Display}} Speed, for {{Display}} Duration.
{{/onTopOfWalking}}
{{#button}}
    /// A press of {{ButtonDisplay}} starts it (within {{Display}} Input Buffer of being able to).
{{/button}}
{{#levelElement}}
    /// It starts within {{Display}} Reach of a {{Element}} in the level (see Level.{{Element}}).
{{/levelElement}}
    /// Replace what it does with your ability: see docs/adding-an-ability.md, which says what each part is for.
    /// </summary>
    [Serializable]
    public class {{Name}} : MovementComponent
    {
        [Tooltip("How long (s) {{Display}} lasts once it starts, at the character's authored size. 0 turns it off")]
        [Min(0f)] [SerializeField] private float {{name}}Duration = 0.5f;
{{#onTopOfWalking}}
        [Tooltip("How fast (m/s) {{Display}} moves the character on the way it faces, on top of walking, at its authored size")]
        [Min(0f)] [SerializeField] private float {{name}}Speed = 4f;
{{/onTopOfWalking}}
{{#button}}
        [Tooltip("If {{ButtonDisplay}} is pressed this many seconds before {{Display}} can start, it starts as soon as it can")]
        [Min(0f)] [SerializeField] private float {{name}}InputBuffer = 0.1f;
{{/button}}
{{#levelElement}}
        [Tooltip("How far (m) from a {{Element}} the character can be, at its authored size, for {{Display}} to start")]
        [Min(0f)] [SerializeField] private float {{name}}Reach = 1.5f;
{{/levelElement}}

        public override bool IsActive(MMotionMotor controller) => controller.{{name}}.IsActive;
{{#takesOver}}
        /// <summary>Under way, it holds the character up (it isn't falling).</summary>
        public override AbilityTraits Traits => AbilityTraits.Supported;
{{/takesOver}}
        /// <summary>
        /// How far through it is (0 to 1), published for its look: an Animator Parameter Translator's float parameter shows
        /// it (Shows > Ability Value > {{Display}} Progress), with no animation code. Add values of your own the same way.
        /// </summary>
        private static readonly AbilityValueId ProgressValue = new("{{Display}} Progress");
        public override string[] AbilityValueNames => abilityValueNames;
        private static readonly string[] abilityValueNames = { ProgressValue.name };

        /// <summary>The events of {{StateFlag}} starting and ending (see MovementComponent.StateEvents).</summary>
        public override StateEventPair[] StateEvents => stateEvents;
        private static readonly StateEventPair[] stateEvents =
        {
            new(CharacterState.{{StateFlag}}, MovementEvents.{{Started}}, MovementEvents.{{Ended}}),
        };

        public override void AdvanceTimers(MovementStep movementData, StepInputs inputs, MMotionMotor controller)
        {
{{#button}}
            controller.{{name}}.press.Track(inputs.{{Button}});
{{/button}}
            controller.{{name}}.timeLeft = MovementRules.CountDown(controller.{{name}}.timeLeft, movementData.deltaTime);
        }

        /// <summary>Ends it at once: another ability took the character, or it was turned off while under way.</summary>
        public override void Interrupt(MovementStep movementData, MMotionMotor controller) => controller.{{name}}.timeLeft = 0f;

{{#takesOver}}
        /// <summary>Carries on while it lasts, or starts if it can, holding the character where it is.</summary>
        public override bool TakeOver(MovementStep movementData, StepInputs inputs, MMotionMotor controller)
        {
            {{Name}}State state = controller.{{name}};
            if (!state.IsActive)
            {
                if (!CanStart(inputs, controller)) { return false; }
                state.timeLeft = {{name}}Duration * controller.DurationScale;
                Mode.InterruptOthers(movementData, controller, this);
            }
            // Your ability's movement goes here. As made: no momentum at all, so the character hovers where it is.
            movementData.velocityOverride = Vector3.zero;
            movementData.state |= CharacterState.{{StateFlag}};
            PublishProgress(controller);
            return true;
        }
{{/takesOver}}
{{#onTopOfWalking}}
        /// <summary>Starts if it can, and while it lasts moves the character on the way it faces, on top of walking.</summary>
        public override void AfterLocomotion(MovementStep movementData, StepInputs inputs, MMotionMotor controller)
        {
            {{Name}}State state = controller.{{name}};
            if (!state.IsActive)
            {
                if (!CanStart(inputs, controller)) { return; }
                state.timeLeft = {{name}}Duration * controller.DurationScale;
            }
            // Your ability's movement goes here. As made: on the way it faces (its physics pose's), at its speed.
            Vector3 forward = movementData.initialRotation * Vector3.forward;
            movementData.directMovementVector += forward * ({{name}}Speed * controller.SpeedScale * movementData.deltaTime);
            movementData.state |= CharacterState.{{StateFlag}};
            PublishProgress(controller);
        }
{{/onTopOfWalking}}

        private void PublishProgress(MMotionMotor controller)
        {
            float duration = {{name}}Duration * controller.DurationScale;
            controller.AbilityValues.Set(ProgressValue, duration > 0f ? 1f - controller.{{name}}.timeLeft / duration : 1f);
        }

        /// <summary>Whether {{Display}} can start this step: turned on by a duration, and what starts it.</summary>
        private bool CanStart(StepInputs inputs, MMotionMotor controller)
        {
            if ({{name}}Duration <= 0f) { return false; }
{{#levelElement}}
            if (!Within{{Name}}Reach(controller)) { return false; }
{{/levelElement}}
{{#button}}
            return controller.{{name}}.press.Consume(inputs.{{Button}}, {{name}}InputBuffer);
{{/button}}
{{^button}}
            return true;
{{/button}}
        }
{{#levelElement}}

        /// <summary>
        /// Whether a {{Element}} is within reach of the character (from its physics pose). Which one doesn't matter here;
        /// an ability that chooses one breaks ties with StableOrder, so it chooses alike on every machine.
        /// </summary>
        private bool Within{{Name}}Reach(MMotionMotor controller)
        {
            float reach = {{name}}Reach * controller.HeightScale;
            Vector3 position = controller.PhysicsPosition;
            // By index: a foreach over the list (an interface) would allocate every step.
            var elements = {{Element}}.All;
            for (int i = 0; i < elements.Count; i++)
            {
                if ((elements[i].transform.position - position).sqrMagnitude <= reach * reach) { return true; }
            }
            return false;
        }
{{/levelElement}}
    }

    /// <summary>What each character is doing with {{Display}} (modes are shared assets, so it lives on the character).</summary>
    public class {{Name}}State
    {
        /// <summary>Seconds left of {{Display}} (0 when it isn't under way).</summary>
        public float timeLeft;
{{#button}}
        /// <summary>{{ButtonDisplay}}'s presses, so each press starts it at most once.</summary>
        public readonly PressBuffer press = new();
{{/button}}

        /// <summary>Whether {{Display}} is under way.</summary>
        public bool IsActive => timeLeft > 0f;
    }

    // This state's place on the character (see MMotionMotor): declared here, beside its ability.
    public partial class MMotionMotor
    {
        /// <summary>The character's {{Display}} state ({{Name}}).</summary>
        public {{Name}}State {{name}} { get; } = new {{Name}}State();
    }
}
